    22→    { slug: "real-estate-sale-abroad", ar: "بيع العقار من الخارج", en: "Property Sale from Abroad", descAr: "بيع العقار الأردني من قبل مالك موجود خارج الأردن.", descEn: "Sale of Jordanian property by an owner located abroad.", parent: "real-estate" },
    23→    { slug: "document-authentication", ar: "تصديق المستندات", en: "Document Authentication", descAr: "تصديق الوثائق الأردنية لاستخدامها في الخارج، أو تصديق الوثائق الأجنبية لاستخدامها في الأردن.", descEn: "Authentication of Jordanian documents for use abroad, or foreign documents for use in Jordan." },
    24→  ];
    25→
    26→  const bySlug: Record<string, { id: string }> = {};
    27→  for (const a of areas) {
    28→    let parentId: string | null = null;
    29→    if (a.parent) {
    30→      const existing = await prisma.practiceArea.findUnique({ where: { slug: a.parent }, select: { id: true } });
    31→      parentId = existing?.id ?? null;
    32→    }
    33→    const created = await prisma.practiceArea.upsert({
    34→      where: { slug: a.slug },
    35→      update: { nameAr: a.ar, nameEn: a.en, descriptionAr: a.descAr, descriptionEn: a.descEn, parentId },
    36→      create: { slug: a.slug, nameAr: a.ar, nameEn: a.en, descriptionAr: a.descAr, descriptionEn: a.descEn, parentId, isActive: true, sortOrder: 0 },
    37→    });
    38→    bySlug[a.slug] = created;
    39→  }
    40→  return bySlug;
    41→}

   408→async function upsertLegalServices(
   409→  paBySlug: Record<string, { id: string }>,
   410→  officialBySlug: Record<string, { id: string }>,
   411→) {
   412→  for (const s of SERVICES) {
   413→    // Look up the practice area — it may already exist in your repo
   414→    const pa = await prisma.practiceArea.findUnique({ where: { slug: s.practiceAreaSlug } });
   415→    const paId = pa?.id ?? paBySlug[s.practiceAreaSlug]?.id ?? null;
   416→
   417→    const created = await prisma.legalService.upsert({
   418→      where: { slug: s.slug },
   419→      update: {
   420→        code: s.code, nameAr: s.nameAr, nameEn: s.nameEn,
   421→        shortAr: s.shortAr, shortEn: s.shortEn,
   422→        descriptionAr: s.descriptionAr, descriptionEn: s.descriptionEn,
   423→        practiceAreaId: paId,
   424→        defaultRemoteEligibility: s.remoteEligibility,
   425→        platformFeeDefault: s.platformFee, lawyerFeeMin: s.lawyerFeeMin, lawyerFeeMax: s.lawyerFeeMax,
   426→        governmentFeeEstimate: s.govtFee,
   427→        governmentFeeNoteAr: s.govtFeeNoteAr ?? null, governmentFeeNoteEn: s.govtFeeNoteEn ?? null,
   428→        typicalDurationDays: s.durationDays, isActive: true, isFeatured: s.isFeatured ?? false, sortOrder: s.sortOrder,
   429→      },
   430→      create: {
   431→        slug: s.slug, code: s.code, nameAr: s.nameAr, nameEn: s.nameEn,
   432→        shortAr: s.shortAr, shortEn: s.shortEn,
   433→        descriptionAr: s.descriptionAr, descriptionEn: s.descriptionEn,
   434→        practiceAreaId: paId,
   435→        defaultRemoteEligibility: s.remoteEligibility,
   436→        platformFeeDefault: s.platformFee, lawyerFeeMin: s.lawyerFeeMin, lawyerFeeMax: s.lawyerFeeMax,
   437→        governmentFeeEstimate: s.govtFee,
   438→        governmentFeeNoteAr: s.govtFeeNoteAr ?? null, governmentFeeNoteEn: s.govtFeeNoteEn ?? null,
   439→        typicalDurationDays: s.durationDays, isActive: true, isFeatured: s.isFeatured ?? false, sortOrder: s.sortOrder,
   440→      },
   441→    });
   442→
   443→    await prisma.legalProcedure.upsert({
   444→      where: { slug: s.procedure.slug },
   445→      update: {
   446→        legalServiceId: created.id, nameAr: s.procedure.nameAr, nameEn: s.procedure.nameEn,
   447→        remoteEligibility: s.procedure.remoteEligibility,
   448→        remoteEligibilityReasonAr: s.procedure.reasonAr, remoteEligibilityReasonEn: s.procedure.reasonEn,
   449→        physicalPresenceSteps: s.procedure.physicalSteps,
   450→        remoteSteps: s.procedure.remoteSteps,
   451→        authorityAr: s.procedure.authorityAr ?? null, authorityEn: s.procedure.authorityEn ?? null,
   452→        legalBasisAr: s.procedure.legalBasisAr ?? null, legalBasisEn: s.procedure.legalBasisEn ?? null,
   453→        notesAr: s.procedure.notesAr ?? null, notesEn: s.procedure.notesEn ?? null,
   454→        estimatedDurationDays: s.procedure.durationDays,
   455→      },
   456→      create: {
   457→        legalServiceId: created.id, slug: s.procedure.slug,
   458→        nameAr: s.procedure.nameAr, nameEn: s.procedure.nameEn,
   459→        remoteEligibility: s.procedure.remoteEligibility,
   460→        remoteEligibilityReasonAr: s.procedure.reasonAr, remoteEligibilityReasonEn: s.procedure.reasonEn,
   461→        physicalPresenceSteps: s.procedure.physicalSteps,
   462→        remoteSteps: s.procedure.remoteSteps,
   463→        authorityAr: s.procedure.authorityAr ?? null, authorityEn: s.procedure.authorityEn ?? null,
   464→        legalBasisAr: s.procedure.legalBasisAr ?? null, legalBasisEn: s.procedure.legalBasisEn ?? null,
   465→        notesAr: s.procedure.notesAr ?? null, notesEn: s.procedure.notesEn ?? null,
   466→        estimatedDurationDays: s.procedure.durationDays,
   467→      },
   468→    });
   469→
   470→    for (const d of s.docs) {
   471→      const existing = await prisma.legalDocumentRequirement.findFirst({
   472→        where: { legalServiceId: created.id, slug: d.slug },
   473→      });
   474→      const payload = {
   475→        legalServiceId: created.id, slug: d.slug,
   476→        nameAr: d.nameAr, nameEn: d.nameEn,
   477→        descriptionAr: d.descAr ?? null, descriptionEn: d.descEn ?? null,
   478→        isRequired: d.isRequired ?? true, provider: d.provider ?? "client", stage: d.stage ?? "at_intake",
   479→        acceptsDigital: d.acceptsDigital ?? true, requiresOriginal: d.requiresOriginal ?? false,
   480→        requiresNotarization: d.requiresNotarization ?? false, requiresApostille: d.requiresApostille ?? false,
   481→      };
   482→      if (existing) await prisma.legalDocumentRequirement.update({ where: { id: existing.id }, data: payload });
   483→      else await prisma.legalDocumentRequirement.create({ data: payload });
   484→    }
   485→
   486→    for (const osSlug of s.officialSourceSlugs) {
   487→      const os = officialBySlug[osSlug];
   488→      if (!os) continue;
   489→      const existing = await prisma.legalServiceOfficialSource.findUnique({
   490→        where: { legalServiceId_officialSourceId: { legalServiceId: created.id, officialSourceId: os.id } },
   491→      });
   492→      if (!existing) {
   493→        await prisma.legalServiceOfficialSource.create({
   494→          data: { legalServiceId: created.id, officialSourceId: os.id, relationType: "primary" },
   495→        });
   496→      }
   497→    }
   498→  }
   499→}
